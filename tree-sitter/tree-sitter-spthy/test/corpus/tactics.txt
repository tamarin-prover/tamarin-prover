==================================================
Presort rejects a pair of rankings
:error
==================================================

theory T begin
tactic: Bad
presort: ss
prio: isFactName "A"
end

---

==================================================
Presort rejects four rankings
:error
==================================================

theory T begin
tactic: Bad
presort: cisp
prio: isFactName "A"
end

---

==================================================
Presort rejects an oracle
:error
==================================================

theory T begin
tactic: Bad
presort: o
prio: isFactName "A"
end

---

==================================================
Presort rejects a tactic reference
:error
==================================================

theory T begin
tactic: Bad
presort: {Other}
prio: isFactName "A"
end

---

==================================================
Priority cannot follow deprioritization
:error
==================================================

theory T begin
tactic: Bad
deprio: isFactName "A"
prio: isFactName "B"
end

---

==================================================
Presort cannot split a ranking from a formal comment identifier
:error
==================================================

theory T begin
tactic: Bad
presort: ss{* hello *}
end

---

==================================================
Presort cannot split a ranking from a priority keyword
:error
==================================================

theory T begin
tactic: Bad
presort: sprio: regex "x"
end

---

==================================================
Presort allows a separate formal comment
==================================================

theory T begin
tactic: Custom
presort: s s{* hello *}
end
